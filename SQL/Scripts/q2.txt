DROP TABLE IF EXISTS OrderItems_Clean;

WITH RankedItems AS (
    SELECT
        OrderItemID,
        OrderID,
        ProductID,
        Quantity,
        UnitPrice,
        LoadedAt,
        ROW_NUMBER() OVER (
            PARTITION BY OrderItemID
            ORDER BY LoadedAt DESC
        ) AS RowNum
    FROM OrderItems
)
SELECT
    OrderItemID,
    OrderID,
    ProductID,
    Quantity,
    UnitPrice,
    LoadedAt
INTO OrderItems_Clean
FROM RankedItems
WHERE RowNum = 1;
SELECT *
FROM OrderItems_Clean
ORDER BY OrderItemID;
